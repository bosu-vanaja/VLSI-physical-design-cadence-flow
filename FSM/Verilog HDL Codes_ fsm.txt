Verilog HDL fsm.v 
 
`timescale 1ns/1ns 
module mealy_1011(input clk, rst, x, output reg y); 
reg [1:0] state, next; 
always @(posedge clk or posedge rst) 
if (rst) state <= 0; 
else state <= next; 
always @(*) begin 
case(state) 
0: next = x ? 1 : 0; 
1: next = x ? 1 : 2; 
2: next = x ? 3 : 0; 
3: next = x ? 1 : 2; 
endcase 
end 
always @(*) begin 
y = (state==3 && x==1); 
end 
endmodule
