Registerbank.v: 
 
`timescale 1ns/1ns 
module reg_bank( 
input clk, rst, we, 
input [2:0] waddr, raddr1, raddr2, 
input [7:0] wdata, 
output [7:0] rdata1, rdata2 
); 
reg [7:0] regs[7:0]; 
integer i; 
always @(posedge clk or posedge rst) begin 
if (rst) 
for(i=0;i<8;i=i+1) regs[i] <= 0; 
else if (we) 
regs[waddr] <= wdata; 
end 
assign rdata1 = regs[raddr1]; 
assign rdata2 = regs[raddr2]; 
endmodule
