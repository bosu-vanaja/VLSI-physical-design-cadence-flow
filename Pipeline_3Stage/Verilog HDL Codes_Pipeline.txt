Pipeline.v: 
 
`timescale 1ns/1ns 
module pipeline(input clk, rst, 
input [7:0] a,b, 
output reg [7:0] y); 
reg [7:0] r1, r2; 
always @(posedge clk or posedge rst) begin 
if (rst) begin 
r1<=0; r2<=0; y<=0; 
end else begin 
r1 <= a + b; // stage 1 
r2 <= r1; // stage 2 
y <= r2; // stage 3 
end 
end 
endmodule

module tb_pipeline; 
reg clk=0, rst; 
reg [7:0] a,b;

endmodule
