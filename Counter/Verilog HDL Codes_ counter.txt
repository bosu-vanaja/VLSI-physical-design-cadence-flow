Verilog HDL counter.v 
`timescale 1ns/1ns 
module counter8(input clk, rst, output reg [7:0] count); 
always @(posedge clk or posedge rst) begin 
if (rst) 
count <= 0; 
else 
count <= count + 1; 
end 
endmodule 
 
Testbench:

`timescale 1ns/1ns 
module tb_counter; 
reg clk=0, rst; 
wire [7:0] count; 
counter8 uut(clk,rst,count); 
always #5 clk=~clk; 
initial begin 
rst=1; #10 rst=0; 
#100 $finish;
end
endmodule
